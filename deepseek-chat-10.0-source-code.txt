package com.deepseek.chat;

import android.app.Application;
import android.os.Build;
import android.util.Log;
import java.io.File;
import java.io.FileOutputStream;
import java.io.PrintWriter;
import java.io.StringWriter;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.Locale;

public class App extends Application {
    private static final String TAG = "DeepSeekApp";
    private static final int VERSION_CODE = 34;

    @Override
    public void onCreate() {
        super.onCreate();
        Thread.UncaughtExceptionHandler previous = Thread.getDefaultUncaughtExceptionHandler();
        Thread.setDefaultUncaughtExceptionHandler(new CrashHandler(previous));
        Log.i(TAG, "Global crash handler installed");
    }

    private class CrashHandler implements Thread.UncaughtExceptionHandler {
        private final Thread.UncaughtExceptionHandler previous;

        CrashHandler(Thread.UncaughtExceptionHandler prev) {
            this.previous = prev;
        }

        @Override
        public void uncaughtException(Thread t, Throwable e) {
            try {
                StringWriter sw = new StringWriter();
                PrintWriter pw = new PrintWriter(sw);
                pw.println("=== DeepSeek Crash Log ===");
                pw.println("Time: " + new SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US)
                        .format(new Date()));
                pw.println("Thread: " + t.getName());
                pw.println("Device: " + Build.MANUFACTURER + " " + Build.MODEL);
                pw.println("Android API: " + Build.VERSION.SDK_INT);
                pw.println("App version: 10.0");
                pw.println("Version code: " + VERSION_CODE);
                pw.println();
                e.printStackTrace(pw);
                pw.flush();

                File dir = getExternalFilesDir(null);
                if (dir == null) dir = getFilesDir();
                File logFile = new File(dir, "deepseek_crash_log.txt");
                FileOutputStream fos = new FileOutputStream(logFile, true);
                fos.write(sw.toString().getBytes("UTF-8"));
                fos.write("\n\n".getBytes("UTF-8"));
                fos.close();
                Log.e(TAG, "Crash log written to: " + logFile.getAbsolutePath());
            } catch (Throwable ignored) {
            }

            if (previous != null) {
                previous.uncaughtException(t, e);
            }
        }
    }
}

    private String postJson(String path, JSONObject body) throws Exception {
        HttpURLConnection conn = (HttpURLConnection) new URL(BASE_URL + path).openConnection();
        conn.setRequestMethod("POST");
        conn.setDoOutput(true);
        conn.setConnectTimeout(30000);
        conn.setReadTimeout(30000);
        Map<String, String> headers = baseHeaders();
        headers.put("Content-Type", "application/json");
        for (Map.Entry<String, String> e : headers.entrySet()) {
            conn.setRequestProperty(e.getKey(), e.getValue());
        }
        byte[] payload = body.toString().getBytes(StandardCharsets.UTF_8);
        OutputStream os = conn.getOutputStream();
        try {
            os.write(payload);
        } finally {
            os.close();
        }
        int code = conn.getResponseCode();
        if (code != 200) {
            String err = readAll(conn.getErrorStream());
            throw new RuntimeException("HTTP " + code + " on " + path + ": "
                    + err.substring(0, Math.min(err.length(), 300)));
        }
        return readAll(conn.getInputStream());
    }

    private int solvePowInternal(JSONObject challenge, StreamCallback cb, String prefix) throws Exception {
        String challengeStr = challenge.getString("challenge");
        String salt = challenge.getString("salt");
        long expireAt = challenge.getLong("expire_at");
        int difficulty = challenge.getInt("difficulty");
        String fullPrefix = salt + "_" + expireAt + "_";
        byte[] target = hexToBytes(challengeStr);

        if (cb != null) {
            StringBuilder msg = new StringBuilder();
            if (prefix != null && !prefix.isEmpty()) msg.append(prefix).append(" ");
            msg.append("PoW (difficulty=").append(difficulty).append(")...");
            cb.onLog(msg.toString());
        }
        try {
            Process.setThreadPriority(10);
        } catch (Throwable ignored) {
        }
        long start = System.currentTimeMillis();
        for (int answer = 0; answer < difficulty; answer++) {
            // Use the string-based hash to match the Python implementation exactly.
            String hashHex = DeepSeekHashV1.hash(fullPrefix + answer);
            byte[] hashBytes = hexToBytes(hashHex);
            if (bytesEqual(hashBytes, target)) {
                long elapsed = (System.currentTimeMillis() - start) / 1000;
                if (cb != null) {
                    String p = (prefix != null && !prefix.isEmpty()) ? prefix + " " : "";
                    cb.onLog(p + "PoW solved in " + elapsed + "s (answer=" + answer + ")");
                }
                return answer;
            }
        }
        throw new RuntimeException("PoW not solved within " + difficulty + " attempts");
    }

    private static String makePowHeader(JSONObject challenge, int answer) throws Exception {
        JSONObject obj = new JSONObject();
        obj.put("algorithm", challenge.optString("algorithm", "DeepSeekHashV1"));
        obj.put("challenge", challenge.getString("challenge"));
        obj.put("salt", challenge.getString("salt"));
        obj.put("answer", answer);
        obj.put("signature", challenge.getString("signature"));
        obj.put("target_path", challenge.optString("target_path", "/api/v0/chat/completion"));
        return Base64.encodeToString(obj.toString().getBytes(StandardCharsets.UTF_8), 2);
    }

    private static byte[] hexToBytes(String hex) {
        byte[] out = new byte[32];
        for (int i = 0; i < 32; i++) {
            int hi = Character.digit(hex.charAt(i * 2), 16);
            int lo = Character.digit(hex.charAt(i * 2 + 1), 16);
            out[i] = (byte) (lo | (hi << 4));
        }
        return out;
    }

    private static boolean bytesEqual(byte[] a, byte[] b) {
        if (a.length != b.length) return false;
        int diff = 0;
        for (int i = 0; i < a.length; i++) diff |= a[i] ^ b[i];
        return diff == 0;
    }

    public FileInfo uploadFile(Attachment attachment, boolean thinking, String modelType,
                                StreamCallback cb) throws Exception {
        String displayName = attachment.displayName;
        if (displayName == null || displayName.isEmpty()) {
            displayName = "upload_" + System.currentTimeMillis() + ".bin";
        }
        String safeName = displayName.replaceAll("[\"\\r\\n]", "_");
        if (attachment.bytes == null || attachment.bytes.length == 0) {
            throw new RuntimeException("File empty: " + safeName);
        }
        JSONObject pow = getPowChallenge("/api/v0/file/upload_file");
        String powHeader = makePowHeader(pow, solvePowInternal(pow, cb, "Upload"));

        String boundary = "----DSB" + System.currentTimeMillis();
        ByteArrayOutputStream body = new ByteArrayOutputStream();
        body.write(("--" + boundary + "\r\n"
                + "Content-Disposition: form-data; name=\"file\"; filename=\""
                + safeName + "\"\r\n"
                + "Content-Type: application/octet-stream\r\n\r\n").getBytes(StandardCharsets.UTF_8));
        body.write(attachment.bytes);
        body.write(("\r\n--" + boundary + "--\r\n").getBytes(StandardCharsets.UTF_8));
        byte[] bodyBytes = body.toByteArray();

        HttpURLConnection conn = (HttpURLConnection) new URL(BASE_URL + "/api/v0/file/upload_file").openConnection();
        conn.setRequestMethod("POST");
        conn.setDoOutput(true);
        conn.setDoInput(true);
        conn.setConnectTimeout(60000);
        conn.setReadTimeout(300000);
        conn.setUseCaches(false);
        Map<String, String> headers = baseHeaders();
        headers.put("Content-Type", "multipart/form-data; boundary=" + boundary);
        headers.put("x-ds-pow-response", powHeader);
        headers.put("x-thinking-enabled", thinking ? "1" : "0");
        headers.put("x-model-type", modelType == null ? "default" : modelType);
        headers.put("x-file-size", String.valueOf(attachment.bytes.length));
        for (Map.Entry<String, String> e : headers.entrySet()) {
            conn.setRequestProperty(e.getKey(), e.getValue());
        }
        conn.setFixedLengthStreamingMode(bodyBytes.length);
        OutputStream os = conn.getOutputStream();
        os.write(bodyBytes);
        os.flush();
        os.close();

        int code = conn.getResponseCode();
        if (code != 200) {
            String err = readAll(conn.getErrorStream());
            throw new RuntimeException("HTTP " + code + ": " + err.substring(0, Math.min(err.length(), 500)));
        }
        String resp = readAll(conn.getInputStream());
        conn.disconnect();

        JSONObject root = new JSONObject(resp);
        if (root.optInt("code", -1) != 0) {
            throw new RuntimeException("Upload failed: " + resp);
        }
        JSONObject data = root.optJSONObject("data");
        if (data == null) throw new RuntimeException("data null: " + resp);
        JSONObject biz = data.optJSONObject("biz_data");
        if (biz != null || !data.has("id")) {
            data = biz;
        }
        if (data == null) throw new RuntimeException("biz_data null: " + resp);

        FileInfo info = new FileInfo();
        info.id = data.optString("id", null);
        info.fileName = data.optString("file_name", safeName);
        info.fileSize = data.optLong("file_size", attachment.bytes.length);
        info.isImage = data.optBoolean("is_image", false);
        info.status = data.optString("status", "PENDING");
        info.width = data.optInt("width", 0);
        info.height = data.optInt("height", 0);
        info.tokenUsage = data.optInt("token_usage", 0);
        if (info.id == null || info.id.isEmpty()) {
            throw new RuntimeException("file_id missing: " + resp);
        }
        return info;
    }

    public FileInfo waitForFileReady(String fileId, int timeoutSec, float intervalSec,
                                      StreamCallback cb) throws Exception {
        long start = System.currentTimeMillis();
        while (System.currentTimeMillis() - start < timeoutSec * 1000L) {
            FileInfo info = fetchFileInfo(fileId);
            if (cb != null) cb.onLog("وضعیت فایل: " + info.status);
            if ("SUCCESS".equals(info.status)) return info;
            if ("FAILED".equals(info.status)) throw new RuntimeException("File failed: " + info);
            try {
                Thread.sleep((long) (1000f * intervalSec));
            } catch (InterruptedException ignored) {
            }
        }
        throw new RuntimeException("File " + fileId + " not ready within " + timeoutSec + "s");
    }

    //
    // The chat() method always sends parent_message_id = the bot's previous reply
    // id, so all messages stay in one chain and the AI remembers the conversation.
    //
    public void chat(final String prompt,
                     final List<String> refFileIds,
                     final String modelType,
                     final Boolean thinkingEnabled,
                     final Boolean searchEnabled,
                     final String explicitSessionId,   // optional override
                     final Long explicitParentId,      // optional override
                     final StreamCallback cb) {
        new Thread(new Runnable() {
            @Override
            public void run() {
                try {
                    // Resolve mode + thinking/search toggles
                    String mode = (modelType == null) ? "default" : modelType;
                    boolean thinking;
                    boolean search;
                    if (thinkingEnabled != null) {
                        thinking = thinkingEnabled;
                    } else {
                        thinking = "vision".equals(mode);
                    }
                    if (searchEnabled != null) {
                        search = searchEnabled;
                    } else {
                        search = "default".equals(mode);
                    }
                    if (!"default".equals(mode) && search) {
                        search = false;
                    }
                    if ("vision".equals(mode) && (refFileIds == null || refFileIds.isEmpty())) {
                        if (cb != null) cb.onLog("Warning: vision mode without ref_file_ids may not work");
                    }
                    if ("expert".equals(mode) && refFileIds != null && !refFileIds.isEmpty()) {
                        if (cb != null) cb.onLog("Warning: file uploads not supported in expert mode, ignoring");
                    }

                    // Session selection — by default we keep the SAME session id across
                    // all chat() calls, so messages stay in one chat thread.
                    String sessionId = explicitSessionId != null ? explicitSessionId : currentSessionId;
                    if (sessionId == null) {
                        sessionId = createChatSession();
                        // New session → reset chain (no parent yet).
                        lastBotMessageId = null;
                        lastUserMessageId = null;
                        currentSessionId = sessionId;
                    }

                    // Chain selection — use explicit override or auto-pick the last bot id.
                    Long parentId = explicitParentId != null ? explicitParentId : lastBotMessageId;

                    // Solve PoW — retry up to 5 times with fresh challenges.
                    String powHeader = null;
                    Exception lastError = null;
                    for (int attempt = 0; attempt < 5; attempt++) {
                        try {
                            JSONObject challenge = getPowChallenge("/api/v0/chat/completion");
                            int answer = solvePowInternal(challenge, cb, attempt == 0 ? "Chat" : "Chat (retry " + attempt + ")");
                            powHeader = makePowHeader(challenge, answer);
                            break;
                        } catch (Exception e) {
                            lastError = e;
                            if (cb != null) cb.onLog("PoW attempt " + (attempt + 1) + " failed, retrying... (" + e.getMessage() + ")");
                        }
                    }
                    if (powHeader == null) {
                        throw lastError != null ? lastError : new RuntimeException("PoW failed after 5 attempts");
                    }

                    JSONObject body = new JSONObject();
                    body.put("chat_session_id", sessionId);
                    body.put("parent_message_id", parentId == null ? JSONObject.NULL : parentId);
                    body.put("model_type", mode);
                    body.put("prompt", prompt);
                    JSONArray arr = new JSONArray();
                    if (refFileIds != null) {
                        for (String id : refFileIds) arr.put(id);
                    }
                    body.put("ref_file_ids", arr);
                    body.put("thinking_enabled", thinking);
                    body.put("search_enabled", search);

    private void dispatchChunk(StreamCallback cb, String fragmentType, String content,
                                boolean inCodeBlock, String codeLang,
                                boolean codeBlockStarted, boolean codeBlockEnded) {
        if (cb == null || content == null || content.isEmpty()) return;
        // Simple fence detection — split content on ``` markers
        // We process content character-by-character to detect fence transitions.
        // (We re-implement a small state machine because chunks may split fences.)
        // For simplicity, we emit the raw content with current code-block state.
        // The MainActivity handles fence parsing across chunks.
        cb.onChunk(new StructuredChunk(
                fragmentType == null ? "RESPONSE" : fragmentType,
                content,
                inCodeBlock,
                codeLang,
                codeBlockStarted,
                codeBlockEnded));
    }

    /**
     * Find the last ASSISTANT message id in a chat (for chaining).
     * Falls back to the last message id of any role.
     * Returns 0 if the chat has no messages.
     */

        List<ChatMessage> msgs = getMessages(chatSessionId);
        for (int i = msgs.size() - 1; i >= 0; i--) {

            if ("ASSISTANT".equals(m.role) && "FINISHED".equals(m.status)) {

        }
        if (!msgs.isEmpty()) return msgs.get(msgs.size() - 1).messageId;
        return 0;
    }

package com.deepseek.chat;

import java.nio.charset.StandardCharsets;

/**
 * DeepSeekHashV1 — Custom Keccak-256 hash (DeepSeek variant).
 *
 * Ported directly from the Python reference implementation in deepseek_client_4.py.
 * Uses 23 rounds (skips round 0), rate = 136 bytes, capacity = 512 bits,
 * output = 256 bits, padding = 0x06 ... 0x80.
 *
 * IMPORTANT: Java's long is 64-bit signed, so (1L << 64) overflows to 1.
 * We use 0xFFFFFFFFFFFFFFFFL (-1L) as the 64-bit mask instead.
 */
public final class DeepSeekHashV1 {

    /** 64-bit mask: all bits set. Equivalent to (1L << 64) - 1 in Python. */
    private static final long MASK64 = 0xFFFFFFFFFFFFFFFFL; // = -1L
    private static final int MASK32 = 0xFFFFFFFF;

    private static final long[] RC = {
        0x0000000000000001L, 0x0000000000008082L, 0x800000000000808AL, 0x8000000080008000L,
        0x000000000000808BL, 0x0000000080000001L, 0x8000000080008081L, 0x8000000000008009L,
        0x000000000000008AL, 0x0000000000000088L, 0x0000000080008009L, 0x000000008000000AL,
        0x000000008000808BL, 0x800000000000008BL, 0x8000000000008089L, 0x8000000000008003L,
        0x8000000000008002L, 0x8000000000000080L, 0x000000000000800AL, 0x800000008000000AL,
        0x8000000080008081L, 0x8000000000008080L, 0x0000000080000001L, 0x8000000080008008L
    };

    private static final int[] JS_V = {10, 7, 11, 17, 18, 3, 5, 16, 8, 21, 24, 4, 15, 23, 19, 13, 12, 2, 20, 14, 22, 9, 6, 1};
    private static final int[] JS_W = {1, 3, 6, 10, 15, 21, 28, 36, 45, 55, 2, 14, 27, 41, 56, 8, 25, 43, 62, 18, 39, 61, 20, 44};

    private static final int RATE_BYTES = 136;
    private static final int OUTPUT_BYTES = 32;
    private static final byte PADDING_BYTE = 0x06;

    private DeepSeekHashV1() {}

    private static long rotl64(long x, int n) {
        n = n & 63;
        if (n == 0) return x;
        return (x << n) | (x >>> (64 - n));
    }

    /**
     * Keccak-f[1600] permutation — DeepSeek variant (23 rounds, skips round 0).
     */
    private static void keccakF1600(long[] state) {
        for (int rnd = 1; rnd < 24; rnd++) {
            // Theta
            long[] C = new long[5];
            for (int x = 0; x < 5; x++) {
                C[x] = state[x] ^ state[x + 5] ^ state[x + 10] ^ state[x + 15] ^ state[x + 20];
            }
            long[] D = new long[5];
            for (int x = 0; x < 5; x++) {
                D[x] = C[(x + 4) % 5] ^ rotl64(C[(x + 1) % 5], 1);
            }
            for (int i = 0; i < 25; i++) {
                state[i] ^= D[i % 5];
            }

            // Rho + Pi (using the JS chase algorithm)
            long W = state[1];
            for (int i = 0; i < 24; i++) {
                int targetLane = JS_V[i];
                int rotation = JS_W[i];
                long saved = state[targetLane];
                state[targetLane] = rotl64(W, rotation);
                W = saved;
            }

            // Chi
            for (int y = 0; y < 5; y++) {
                int base = y * 5;
                long t0 = state[base];
                long t1 = state[base + 1];
                long t2 = state[base + 2];
                long t3 = state[base + 3];
                long t4 = state[base + 4];
                state[base]     = t0 ^ ((~t1) & t2);
                state[base + 1] = t1 ^ ((~t2) & t3);
                state[base + 2] = t2 ^ ((~t3) & t4);
                state[base + 3] = t3 ^ ((~t4) & t0);
                state[base + 4] = t4 ^ ((~t0) & t1);
            }

            // Iota
            state[0] ^= RC[rnd];
        }
    }

    /**
     * Compute the hash of a message string. Returns a 64-char hex string.
     */
    public static String hash(String message) {
        return hash(message.getBytes(StandardCharsets.UTF_8));
    }

    /**
     * Compute the hash of a message byte array. Returns a 64-char hex string.
     */
    public static String hash(byte[] message) {
        byte[] padded = padMessage(message);
        long[] state = new long[25];

        // Absorb
        for (int blockStart = 0; blockStart < padded.length; blockStart += RATE_BYTES) {
            for (int laneIdx = 0; laneIdx < RATE_BYTES / 8; laneIdx++) {
                int r = blockStart + laneIdx * 8;
                long high32 = ((long)(padded[r + 7] & 0xFF) << 24)
                            | ((long)(padded[r + 6] & 0xFF) << 16)
                            | ((long)(padded[r + 5] & 0xFF) << 8)
                            | ((long)(padded[r + 4] & 0xFF));
                long low32 = ((long)(padded[r + 3] & 0xFF) << 24)
                           | ((long)(padded[r + 2] & 0xFF) << 16)
                           | ((long)(padded[r + 1] & 0xFF) << 8)
                           | ((long)(padded[r] & 0xFF));
                long lane = (high32 << 32) | low32;
                state[laneIdx] ^= lane;
                // No mask needed — Java long is already 64-bit and XOR preserves it.
            }
            keccakF1600(state);
        }

        // Squeeze
        byte[] output = new byte[OUTPUT_BYTES];
        for (int laneIdx = 0; laneIdx < OUTPUT_BYTES / 8; laneIdx++) {
            long high32 = (state[laneIdx] >>> 32) & MASK32;
            long low32 = state[laneIdx] & MASK32;
            int off = laneIdx * 8;
            output[off]     = (byte)(low32 & 0xFF);
            output[off + 1] = (byte)((low32 >> 8) & 0xFF);
            output[off + 2] = (byte)((low32 >> 16) & 0xFF);
            output[off + 3] = (byte)((low32 >> 24) & 0xFF);
            output[off + 4] = (byte)(high32 & 0xFF);
            output[off + 5] = (byte)((high32 >> 8) & 0xFF);
            output[off + 6] = (byte)((high32 >> 16) & 0xFF);
            output[off + 7] = (byte)((high32 >> 24) & 0xFF);
        }
        return toHex(output);
    }

    /**
     * Compute the hash of (base + decimal_string_of_i). Returns 32 raw bytes.
     */
    public static byte[] hashBytes(byte[] base, int i) {
        String fullMessage = new String(base, StandardCharsets.UTF_8) + String.valueOf(i);
        byte[] msgBytes = fullMessage.getBytes(StandardCharsets.UTF_8);
        byte[] padded = padMessage(msgBytes);
        long[] state = new long[25];

        // Absorb
        for (int blockStart = 0; blockStart < padded.length; blockStart += RATE_BYTES) {
            for (int laneIdx = 0; laneIdx < RATE_BYTES / 8; laneIdx++) {
                int r = blockStart + laneIdx * 8;
                long high32 = ((long)(padded[r + 7] & 0xFF) << 24)
                            | ((long)(padded[r + 6] & 0xFF) << 16)
                            | ((long)(padded[r + 5] & 0xFF) << 8)
                            | ((long)(padded[r + 4] & 0xFF));
                long low32 = ((long)(padded[r + 3] & 0xFF) << 24)
                           | ((long)(padded[r + 2] & 0xFF) << 16)
                           | ((long)(padded[r + 1] & 0xFF) << 8)
                           | ((long)(padded[r] & 0xFF));
                long lane = (high32 << 32) | low32;
                state[laneIdx] ^= lane;
            }
            keccakF1600(state);
        }

        // Squeeze
        byte[] output = new byte[OUTPUT_BYTES];
        for (int laneIdx = 0; laneIdx < OUTPUT_BYTES / 8; laneIdx++) {
            long high32 = (state[laneIdx] >>> 32) & MASK32;
            long low32 = state[laneIdx] & MASK32;
            int off = laneIdx * 8;
            output[off]     = (byte)(low32 & 0xFF);
            output[off + 1] = (byte)((low32 >> 8) & 0xFF);
            output[off + 2] = (byte)((low32 >> 16) & 0xFF);
            output[off + 3] = (byte)((low32 >> 24) & 0xFF);
            output[off + 4] = (byte)(high32 & 0xFF);
            output[off + 5] = (byte)((high32 >> 8) & 0xFF);
            output[off + 6] = (byte)((high32 >> 16) & 0xFF);
            output[off + 7] = (byte)((high32 >> 24) & 0xFF);
        }
        return output;
    }

    /**
     * Pad a message with Keccak padding: append 0x06, then zeros, then XOR last byte with 0x80.
     */
    private static byte[] padMessage(byte[] message) {
        int paddedLen = message.length + 1;
        while (paddedLen % RATE_BYTES != 0) paddedLen++;
        byte[] padded = new byte[paddedLen];
        System.arraycopy(message, 0, padded, 0, message.length);
        padded[message.length] = PADDING_BYTE;
        padded[padded.length - 1] ^= (byte)0x80;
        return padded;
    }

    private static final char[] HEX = "0123456789abcdef".toCharArray();

    public static String toHex(byte[] bytes) {
        char[] chars = new char[bytes.length * 2];
        for (int i = 0; i < bytes.length; i++) {
            int b = bytes[i] & 0xFF;
            chars[i * 2] = HEX[b >>> 4];
            chars[i * 2 + 1] = HEX[b & 0x0F];
        }
        return new String(chars);
    }

    /**
     * Continue an existing chat session. Loads its history (so the user sees the
     * previous messages on screen) and primes the client so the next outgoing
     * message will be chained onto the last bot reply.
     */
    private void continueExistingChat(final String sessionId, final String title) {
        if (client == null) {
            String token = tokenInput.getText().toString().trim();
            if (token.isEmpty()) {
                toast("ابتدا توکن را وارد کنید");
                return;
            }
            client = new DeepSeekClient(token, true);
            currentToken = token;
        }
        // Clear UI and show a loading marker.
        chatContainer.removeAllViews();
        thinkingLog.setLength(0);
        thinkingView.setText("");
        thinkingContainer.setVisibility(View.GONE);
        currentTextView = null;
        currentTextBuffer.setLength(0);
        showStatus("بارگذاری چت…");

        final DeepSeekClient c = client;
        new Thread(() -> {
            try {
                final List<DeepSeekClient.ChatMessage> msgs = c.getMessages(sessionId);
                // Prime the client for the next outgoing message:
                //  - use this session id (so we don't create a new one)
                //  - set the last bot message id (so the next message chains
                //    onto the existing conversation, giving AI full memory)
                c.setSessionId(sessionId);
                long lastBot = c.findLastBotMessageId(sessionId);
                c.resetChain();
                c.setLastBotMessageId(lastBot > 0 ? Long.valueOf(lastBot) : null);
                mainHandler.post(() -> {
                    hideStatus();
                    renderHistoryMessages(msgs);
                    toast("ادامه چت: " + (title != null ? title : ""));
                });
            } catch (Exception e) {
                mainHandler.post(() -> {
                    hideStatus();
                    toast("خطا در بارگذاری چت: " + e.getMessage());
                });
            }
        }).start();
    }

    /**
     * Render a historical bot message. Splits on ``` fences so that HTML
     * code blocks get the same Preview button as live messages.
     */
    private void renderHistoryBotMessage(String content) {
        int i = 0;
        boolean inCode = false;
        StringBuilder text = new StringBuilder();
        StringBuilder code = new StringBuilder();
        String lang = null;
        while (i < content.length()) {
            if (!inCode) {
                int fence = content.indexOf("```", i);
                if (fence < 0) {
                    text.append(content, i, content.length());
                    break;
                }
                text.append(content, i, fence);
                if (text.length() > 0) {
                    appendToResponse(text.toString());
                    text.setLength(0);
                }
                int nl = content.indexOf('\n', fence);
                if (nl < 0) {
                    lang = content.substring(fence + 3).trim();
                    i = content.length();
                    inCode = true;
                    break;
                }
                lang = content.substring(fence + 3, nl).trim();
                i = nl + 1;
                inCode = true;
            } else {
                int fence = content.indexOf("```", i);
                if (fence < 0) {
                    code.append(content, i, content.length());
                    break;
                }
                code.append(content, i, fence);
                addCodeBlockCard(lang, code.toString());
                code.setLength(0);
                lang = null;
                inCode = false;
                i = fence + 3;
                if (i < content.length() && content.charAt(i) == '\n') i++;
            }
        }
        if (inCode && code.length() > 0) {
            addCodeBlockCard(lang, code.toString());
        } else if (text.length() > 0) {
            appendToResponse(text.toString());
        }
        finalizeResponseTextView();
    }

    @Override
    protected void onActivityResult(int requestCode, int resultCode, Intent data) {
        super.onActivityResult(requestCode, resultCode, data);

        // Handle "continue chat" coming back from ChatListActivity.
        if (requestCode == REQ_OPEN_CHAT_LIST && resultCode == RESULT_OK && data != null) {
            String sessionId = data.getStringExtra("continue_session_id");
            String title = data.getStringExtra("title");
            if (sessionId != null && !sessionId.isEmpty()) {
                continueExistingChat(sessionId, title);
            }
            return;
        }

        if (resultCode != RESULT_OK || data == null || data.getData() == null) return;
        try {
            Uri uri = data.getData();
            try {
                getContentResolver().takePersistableUriPermission(uri, Intent.FLAG_GRANT_READ_URI_PERMISSION);
            } catch (Throwable ignored) {
            }
            DeepSeekClient.Attachment att = loadAttachmentFromUri(uri);
            if (att == null) {
                toast("خطا در خواندن فایل");
                return;
            }
            pendingAttachments.add(att);
            refreshAttachmentsUI();
            toast("فایل اضافه شد: " + att.displayName);
            sendButton.setEnabled(true);
        } catch (Throwable t) {
            toast("خطا: " + t.getMessage());
        }
    }

    private DeepSeekClient.Attachment loadAttachmentFromUri(Uri uri) {
        try {
            ContentResolver cr = getContentResolver();
            String name = queryDisplayName(uri);
            if (name == null || name.isEmpty()) name = "file_" + System.currentTimeMillis();
            String mime = cr.getType(uri);
            if (mime == null) {
                String ext = MimeTypeMap.getFileExtensionFromUrl(uri.toString());
                if (ext != null && !ext.isEmpty()) {
                    mime = MimeTypeMap.getSingleton().getMimeTypeFromExtension(ext);
                }
            }
            if (mime == null) mime = "application/octet-stream";
            ByteArrayOutputStream baos = new ByteArrayOutputStream();
            InputStream is = cr.openInputStream(uri);
            byte[] buf = new byte[16384];
            int n;
            while ((n = is.read(buf)) > 0) baos.write(buf, 0, n);
            is.close();
            return new DeepSeekClient.Attachment(name, mime, baos.toByteArray());
        } catch (Exception e) {
            return null;
        }
    }

    private void saveCodeToFile(String lang, String code) {
        String ext = "txt";
        if (lang != null) {
            String l = lang.toLowerCase();
            if (l.contains("html")) ext = "html";
            else if (l.contains("python") || l.contains("py")) ext = "py";
            else if (l.contains("javascript") || l.contains("js")) ext = "js";
            else if (l.contains("json")) ext = "json";
            else if (l.contains("css")) ext = "css";
            else if (l.contains("java")) ext = "java";
            else if (l.contains("xml")) ext = "xml";
            else if (l.contains("sql")) ext = "sql";
            else if (l.contains("bash") || l.contains("sh")) ext = "sh";
            else if (l.contains("markdown") || l.contains("md")) ext = "md";
        }
        String fileName = "deepseek_code_" + System.currentTimeMillis() + "." + ext;
        try {
            if (Build.VERSION.SDK_INT >= 29) {
                ContentResolver cr = getContentResolver();
                ContentValues cv = new ContentValues();
                cv.put("_display_name", fileName);
                cv.put("mime_type", getMimeTypeForExt(ext));
                cv.put("relative_path", Environment.DIRECTORY_DOWNLOADS + "/DeepSeek");
                Uri uri = cr.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, cv);
                if (uri != null) {
                    OutputStream os = cr.openOutputStream(uri);
                    os.write(code.getBytes("UTF-8"));
                    os.close();
                    toast("ذخیره شد: " + fileName + "\nدر: Downloads/DeepSeek");
                    return;
                }
            }
            File dir = new File(Environment.getExternalStoragePublicDirectory(
                    Environment.DIRECTORY_DOWNLOADS), "DeepSeek");
            if (!dir.exists()) dir.mkdirs();
            FileOutputStream fos = new FileOutputStream(new File(dir, fileName));
            fos.write(code.getBytes("UTF-8"));
            fos.close();
            toast("ذخیره شد: " + fileName + "\nدر: " + dir.getAbsolutePath());
        } catch (Exception e) {
            try {
                File dir = new File(getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS), "DeepSeek");
                if (!dir.exists()) dir.mkdirs();
                FileOutputStream fos = new FileOutputStream(new File(dir, fileName));
                fos.write(code.getBytes("UTF-8"));
                fos.close();
                toast("ذخیره شد (پوشه برنامه): " + fileName);
            } catch (Exception e2) {
                toast("خطا در ذخیره: " + e2.getMessage());
            }
        }
    }

    private String getMimeTypeForExt(String ext) {
        if (ext == null) return "text/plain";
        String e = ext.toLowerCase();
        switch (e) {
            case "html": case "htm": return "text/html";
            case "js": return "application/javascript";
            case "json": return "application/json";
            case "xml": return "application/xml";
            case "css": return "text/css";
            case "md": return "text/markdown";
            case "py": return "text/x-python";
            case "java": return "text/x-java";
            case "sh": return "application/x-sh";
            case "sql": return "application/sql";
            default: return "text/plain";
        }
    }

    private void doChat(String prompt, List<String> fileIds, String mode, Boolean thinking, Boolean search) {
        // Reset per-response code-block parser state
        inCodeBlock = false;
        codeLang = null;
        codeBuf.setLength(0);
        textBuf.setLength(0);

        final boolean[] firstChunk = {true};

        client.chat(prompt, fileIds, mode, thinking, search, null, null, new DeepSeekClient.StreamCallback() {
            @Override
            public void onChunk(DeepSeekClient.StructuredChunk chunk) {
                mainHandler.post(() -> {
                    try {
                        if (firstChunk[0]) {
                            stopTypingAnimation();
                            hideStatus();
                            firstChunk[0] = false;
                        }
                        if ("THINK".equals(chunk.fragmentType)) {
                            thinkingLog.append(chunk.content);
                            thinkingView.setText(thinkingLog.toString());
                            if (thinkingContainer.getVisibility() != View.VISIBLE) {
                                thinkingContainer.setVisibility(View.VISIBLE);
                            }
                            return;
                        }
                        // Parse markdown ``` fences across chunks.
                        processChunk(chunk.content);
                    } catch (Throwable ignored) {
                    }
                });
            }

            @Override
            public void onComplete(String fullText, String sessionId, String parentMessageId) {
                mainHandler.post(() -> {
                    try {
                        stopTypingAnimation();
                        hideStatus();
                        // Flush any pending code block.
                        if (inCodeBlock) {
                            addCodeBlockCard(codeLang, codeBuf.toString());
                            codeBuf.setLength(0);
                            inCodeBlock = false;
                            codeLang = null;
                        }
                        if (textBuf.length() > 0) {
                            appendToResponse(textBuf.toString());
                            textBuf.setLength(0);
                        }
                        finalizeResponseTextView();
                        resetUI();
                        if (!userScrolling) scrollToBottom();
                    } catch (Throwable ignored) {
                    }
                });
            }

            @Override
            public void onError(String err) {
                mainHandler.post(() -> {
                    try {
                        stopTypingAnimation();
                        hideStatus();
                        startResponseTextView().setText("خطا: " + err);
                        finalizeResponseTextView();
                        resetUI();
                        if (!userScrolling) scrollToBottom();
                    } catch (Throwable ignored) {
                    }
                });
            }

            @Override
            public void onLog(String msg) {
                mainHandler.post(() -> {
                    try {
                        statusView.setText(msg);
                    } catch (Throwable ignored) {
                    }
                });
            }
        });
    }

    /**
     * Parse ```lang ... ``` fences across streamed chunks.
     * Buffer text inside code blocks; emit code-block cards on fence close.
     */
    private void processChunk(String content) {
        int i = 0;
        while (i < content.length()) {
            if (!inCodeBlock) {
                int fence = content.indexOf("```", i);
                if (fence < 0) {
                    textBuf.append(content, i, content.length());
                    appendToResponse(textBuf.toString());
                    textBuf.setLength(0);
                    return;
                }
                textBuf.append(content, i, fence);
                if (textBuf.length() > 0) {
                    appendToResponse(textBuf.toString());
                    textBuf.setLength(0);
                }
                // find end of line after ```
                int lineEnd = content.indexOf('\n', fence);
                if (lineEnd < 0) {
                    // lang on next chunk; just consume ``` for now
                    codeLang = "";
                    i = fence + 3;
                    inCodeBlock = true;
                    return;
                }
                codeLang = content.substring(fence + 3, lineEnd).trim();
                i = lineEnd + 1;
                inCodeBlock = true;
            } else {
                int fence = content.indexOf("```", i);
                if (fence < 0) {
                    codeBuf.append(content, i, content.length());
                    return;
                }
                codeBuf.append(content, i, fence);
                addCodeBlockCard(codeLang, codeBuf.toString());
                codeBuf.setLength(0);
                codeLang = null;
                inCodeBlock = false;
                i = fence + 3;
                // skip trailing newline if any
                if (i < content.length() && content.charAt(i) == '\n') i++;
            }
        }
    }

                        .setMessage("تغییرات ذخیره نشده از بین برود؟")
                        .setPositiveButton("بله", (d, w) -> resetEditor())
                        .setNegativeButton("انصراف", null)
                        .show();
            } else {
                resetEditor();
            }
        });

        contentEdit.addTextChangedListener(new TextWatcher() {
            @Override public void beforeTextChanged(CharSequence s, int a, int b, int c) {}
            @Override public void onTextChanged(CharSequence s, int a, int b, int c) {}
            @Override
            public void afterTextChanged(Editable e) {
                dirty = true;
                statusLabel.setText("ویرایش نشده*");
            }
        });

        // Default extension
        extEdit.setText("txt");
        // Ensure app directory exists

    private void resetEditor() {
        contentEdit.setText("");
        nameEdit.setText("");
        extEdit.setText("txt");
        currentOpenedUri = null;
        currentFileName = null;
        dirty = false;
        statusLabel.setText("فایل جدید");
        updatePathLabel("downloads/" + APP_DIR_NAME + "/");

    private void ensureAppDir() {
        try {
            if (Build.VERSION.SDK_INT < 29) {
                File downloads = Environment.getExternalStoragePublicDirectory(
                        Environment.DIRECTORY_DOWNLOADS);
                if (downloads != null) {
                    File appDir = new File(downloads, APP_DIR_NAME);
                    if (!appDir.exists()) appDir.mkdirs();
                }
            }
        } catch (Throwable ignored) {
        }
    }

    private File getAppDirLegacy() {
        File downloads = Environment.getExternalStoragePublicDirectory(
                Environment.DIRECTORY_DOWNLOADS);
        File dir = new File(downloads, APP_DIR_NAME);
        if (!dir.exists()) dir.mkdirs();
        return dir;
    }

    private void pickFile() {
        if (dirty) {
            new AlertDialog.Builder(this)
                    .setTitle("باز کردن فایل")
                    .setMessage("تغییرات ذخیره نشده از بین برود؟")
                    .setPositiveButton("بله", (d, w) -> doPickFile())
                    .setNegativeButton("انصراف", null)
                    .show();
        } else {
            doPickFile();
        }
    }

    private void doPickFile() {
        try {
            Intent intent = new Intent(Intent.ACTION_GET_CONTENT);
            intent.setType("*/*");
            intent.addCategory(Intent.CATEGORY_OPENABLE);
            startActivityForResult(Intent.createChooser(intent, "باز کردن فایل متنی"), REQ_OPEN_FILE);
        } catch (Exception e) {
            toast("مرورگر فایل در دسترس نیست");
        }
    }

    @Override
    protected void onActivityResult(int requestCode, int resultCode, Intent data) {
        super.onActivityResult(requestCode, resultCode, data);
        if (requestCode != REQ_OPEN_FILE) return;
        if (resultCode != RESULT_OK || data == null || data.getData() == null) return;
        Uri uri = data.getData();
        try {
            getContentResolver().takePersistableUriPermission(uri,
                    Intent.FLAG_GRANT_READ_URI_PERMISSION);
        } catch (Throwable ignored) {
        }
        loadFromFile(uri);
    }

    private void loadFromFile(Uri uri) {
        try {
            ContentResolver cr = getContentResolver();
            String name = queryDisplayName(uri);
            InputStream is = cr.openInputStream(uri);
            BufferedReader reader = new BufferedReader(new InputStreamReader(is, StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            String line;
            boolean first = true;
            while ((line = reader.readLine()) != null) {
                if (!first) sb.append("\n");
                sb.append(line);
                first = false;
            }
            reader.close();
            is.close();
            contentEdit.setText(sb.toString());
            currentOpenedUri = uri.toString();
            currentFileName = name;

            // Split name + extension
            if (name != null) {
                int dot = name.lastIndexOf('.');
                if (dot > 0) {
                    nameEdit.setText(name.substring(0, dot));
                    extEdit.setText(name.substring(dot + 1));
                } else {
                    nameEdit.setText(name);
                    extEdit.setText("txt");
                }
            }
            dirty = false;
            statusLabel.setText("باز شد: " + name);
            updatePathLabel("باز شده از حافظه");
            toast("فایل بارگذاری شد");
        } catch (Exception e) {
            toast("خطا در باز کردن: " + e.getMessage());
        }
    }

    private String queryDisplayName(Uri uri) {
        try (android.database.Cursor c = getContentResolver().query(
                uri, new String[]{OpenableColumns.DISPLAY_NAME}, null, null, null)) {
            if (c != null && c.moveToFirst()) return c.getString(0);
        } catch (Exception ignored) {
        }
        return uri.getLastPathSegment();
    }

    private void saveFile() {
        String name = nameEdit.getText().toString().trim();
        String ext = extEdit.getText().toString().trim().toLowerCase();
        String content = contentEdit.getText().toString();
        if (name.isEmpty()) {
            toast("نام فایل را وارد کنید");
            return;
        }
        if (ext.isEmpty()) ext = "txt";
        // Sanitize
        name = name.replaceAll("[/\\\\:*?\"<>|]", "_");
        ext = ext.replaceAll("[/\\\\:*?\"<>|.]", "");
        String fileName = name + "." + ext;
        String mime = mimeForExt(ext);
        byte[] bytes = content.getBytes(StandardCharsets.UTF_8);

        try {
            if (Build.VERSION.SDK_INT >= 29) {
                ContentResolver cr = getContentResolver();
                ContentValues cv = new ContentValues();
                cv.put("_display_name", fileName);
                cv.put("mime_type", mime);
                cv.put("relative_path", Environment.DIRECTORY_DOWNLOADS + "/" + APP_DIR_NAME);
                Uri existing = findExistingFile(cr, fileName);
                Uri uri;
                if (existing != null) {
                    // Overwrite
                    OutputStream os = cr.openOutputStream(existing, "w");
                    os.write(bytes);
                    os.close();
                    uri = existing;
                } else {
                    uri = cr.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, cv);
                    if (uri == null) throw new RuntimeException("MediaStore insert failed");
                    OutputStream os = cr.openOutputStream(uri);
                    os.write(bytes);
                    os.close();
                }
                currentOpenedUri = uri.toString();
                currentFileName = fileName;
                dirty = false;
                statusLabel.setText("ذخیره شد: " + fileName);
                updatePathLabel("Downloads/" + APP_DIR_NAME + "/" + fileName);
                toast("ذخیره شد: " + fileName);
            } else {
                File dir = getAppDirLegacy();
                File file = new File(dir, fileName);
                FileOutputStream fos = new FileOutputStream(file);
                fos.write(bytes);
                fos.close();
                currentOpenedUri = Uri.fromFile(file).toString();
                currentFileName = fileName;
                dirty = false;
                statusLabel.setText("ذخیره شد: " + fileName);
                updatePathLabel(file.getAbsolutePath());
                toast("ذخیره شد: " + fileName);
            }
        } catch (Exception e) {
            // Fallback to app-private external dir
            try {
                File dir = new File(getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS), APP_DIR_NAME);
                if (!dir.exists()) dir.mkdirs();
                File file = new File(dir, fileName);
                FileOutputStream fos = new FileOutputStream(file);
                fos.write(bytes);
                fos.close();
                currentOpenedUri = Uri.fromFile(file).toString();
                currentFileName = fileName;
                dirty = false;
                statusLabel.setText("ذخیره شد (پوشه برنامه): " + fileName);
                updatePathLabel(file.getAbsolutePath());
                toast("ذخیره شد (پوشه برنامه): " + fileName);
            } catch (Exception e2) {
                toast("خطا در ذخیره: " + e2.getMessage());
            }
        }
    }

    private Uri findExistingFile(ContentResolver cr, String displayName) {
        try {
            android.database.Cursor c = cr.query(
                    MediaStore.Downloads.EXTERNAL_CONTENT_URI,
                    new String[]{MediaStore.MediaColumns._ID},
                    "_display_name = ? AND relative_path LIKE ?",
                    new String[]{displayName, "%" + APP_DIR_NAME + "%"},
                    null);
            if (c != null && c.moveToFirst()) {
                long id = c.getLong(0);
                c.close();
                return Uri.withAppendedPath(
                        MediaStore.Downloads.EXTERNAL_CONTENT_URI, String.valueOf(id));
            }
            if (c != null) c.close();
        } catch (Exception ignored) {
        }
        return null;
    }

    private String mimeForExt(String ext) {
        switch (ext.toLowerCase()) {
            case "html": case "htm": return "text/html";
            case "css": return "text/css";
            case "js": return "application/javascript";
            case "json": return "application/json";
            case "xml": return "application/xml";
            case "md": return "text/markdown";
            case "py": return "text/x-python";
            case "java": return "text/x-java";
            case "csv": return "text/csv";
            case "txt": default: return "text/plain";
        }
    }

    private void updatePathLabel(String path) {
        pathLabel.setText(path);
    }

    private void toast(String text) {
        try {
            Toast.makeText(this, text, Toast.LENGTH_SHORT).show();
        } catch (Throwable ignored) {
        }
    }

    private int dp(int v) {
        return (int) TypedValue.applyDimension(TypedValue.COMPLEX_UNIT_DIP, v,
                getResources().getDisplayMetrics());
    }
}
